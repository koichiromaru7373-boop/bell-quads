create or replace view public.v_past_log as select h.id::text as id, h.created_at, h.staff_name, h.action as action, h.user_id::text as target_user_id, h.user_name as target_user_name, h.amount as amount, jsonb_build_object('type', h.type, 'bqrp', h.bqrp) as detail, 'history' as source from history h union all select l.id::text as id, l.created_at, l.staff_name, case when l.delta > 0 then 'チケット付与' else 'チケット使用' end as action, l.user_id::text as target_user_id, l.user_name as target_user_name, l.delta as amount, jsonb_build_object('reason', l.reason, 'type_id', l.type_id, 'expires_at', l.expires_at) as detail, 'ticket_ledger' as source from ticket_ledger l; grant select on public.v_past_log to anon, authenticated;
